#  이미지는 크기가 같아서 stack 가능
#  박스는 개수 다르므로 리스트로 담아야 함 → collate_fn이 하는 일
#  이 패턴이 그대로 SSD, YOLO 등에서 사용됨
# [
#   (img1, 3개의 boxes, 3개의 labels),
#   (img2, 7개의 boxes, 7개의 labels)
# ]
#



import torch
from torch.utils.data import Dataset, DataLoader
import random

# -----------------------------
# 1. 아주 간단한 가짜 Dataset
# -----------------------------
class SimpleDataset(Dataset):
    def __len__(self):
        return 4  # 4개의 데이터

    def __getitem__(self, idx):
        # (C,H,W) 텐서 이미지
        img = torch.randn(3, 224, 224)

        # 랜덤 박스 개수 (1~5개)
        num_boxes = random.randint(1, 5)

        # (x1,y1,x2,y2) 4개 좌표
        boxes = torch.randn(num_boxes, 4)

        # 라벨도 박스 개수만큼
        labels = torch.randint(0, 3, (num_boxes,))

        return img, boxes, labels



def simple_collate_fn(batch):
    images = []
    boxes = []
    labels = []

    for img, box, label in batch:
        images.append(img)
        boxes.append(box)
        labels.append(label)

    return torch.stack(images), boxes, labels



dataset = SimpleDataset()
loader = DataLoader(dataset, batch_size=2, collate_fn=simple_collate_fn)

for imgs, boxes, labels in loader:
    print("이미지 shape:", imgs.shape)
    print("박스 개수:", [b.shape for b in boxes])
    print("라벨 개수:", [l.shape for l in labels])
    print("---")
